def create_directory_structure():
    """Create necessary directories for the project."""
    os.makedirs("static/uploads", exist_ok=True)
    os.makedirs("model", exist_ok=True)
    os.makedirs("dataset/processed", exist_ok=True)

def split_dataset(input_folder="dataset/cell_images", output_folder="dataset/processed", train_ratio=0.8, val_ratio=0.1, test_ratio=0.1):
    """Split the dataset into train, validation and test sets using splitfolders."""
    print("🔄 Splitting dataset...")
    splitfolders.ratio(input_folder, output=output_folder, ratio=(train_ratio, val_ratio, test_ratio), seed=42)

    print("✅ Dataset split complete.")
    return (
        os.path.join(output_folder, "train"),
        os.path.join(output_folder, "val"),
        os.path.join(output_folder, "test")
    )

def count_images(folder):
    """Count the total number of image files in a folder (recursively)."""
    return sum(len([f for f in files if f.lower().endswith(('.png', '.jpg', '.jpeg'))]) for _, _, files in os.walk(folder))

def create_data_generators(train_folder, val_folder, test_folder, img_size=(224, 224), batch_size=32):
    """Create training, validation, and test data generators with preprocessing and augmentation."""
    
    print("⚙ Creating data generators...")
    
    train_datagen = ImageDataGenerator(
        rescale=1./255,
        rotation_range=40,
        width_shift_range=0.2,
        height_shift_range=0.2,
        shear_range=0.2,
        zoom_range=0.2,
        horizontal_flip=True,
        fill_mode='nearest'
    )
    
    val_test_datagen = ImageDataGenerator(rescale=1./255)
    
    train_generator = train_datagen.flow_from_directory(
        train_folder,
        target_size=img_size,
        batch_size=batch_size,
        class_mode='binary',
        color_mode='rgb',
        shuffle=True
    )
    
    val_generator = val_test_datagen.flow_from_directory(
        val_folder,
        target_size=img_size,
        batch_size=batch_size,
        class_mode='binary',
        color_mode='rgb',
        shuffle=False
    )
    
    test_generator = val_test_datagen.flow_from_directory(
        test_folder,
        target_size=img_size,
        batch_size=batch_size,
        class_mode='binary',
        color_mode='rgb',
        shuffle=False
    )
    
    print("✅ Generators ready.")
    return train_generator, val_generator, test_generator

def visualize_samples(folder, num_samples=4, classes=["Parasitized", "Uninfected"]):
    """Visualize and save sample images from the training dataset."""
    plt.figure(figsize=(12, 6))

    for idx, class_name in enumerate(classes):
        class_path = os.path.join(folder, class_name)
        images = os.listdir(class_path)[:num_samples]

        for i, img_name in enumerate(images):
            img_path = os.path.join(class_path, img_name)
            img = cv2.imread(img_path)
            if img is None:
                continue
            img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)

            plt.subplot(len(classes), num_samples, idx * num_samples + i + 1)
            plt.imshow(img)
            plt.title(class_name)
            plt.axis("off")

    plt.tight_layout()
    plt.savefig("static/sample_images.png")
    plt.close()
    print("🖼 Saved visualization to static/sample_images.png")

if __name__== "__main__":
    create_directory_structure()
    train, val, test = split_dataset()
    create_data_generators(train, val, test)
    visualize_samples(train)

import os
import numpy as np
import matplotlib.pyplot as plt
from tensorflow.keras.models import Sequential # type: ignore
from tensorflow.keras.layers import Input, Conv2D, MaxPooling2D, Flatten, Dense, Dropout, BatchNormalization # type: ignore
from tensorflow.keras.callbacks import EarlyStopping, ModelCheckpoint, ReduceLROnPlateau
from tensorflow.keras.optimizers import Adam
from preprocess import split_dataset, create_data_generators
from sklearn.utils.class_weight import compute_class_weight
from sklearn.metrics import confusion_matrix, classification_report
import seaborn as sns

# Constants
EPOCHS = 30
BATCH_SIZE = 16
IMG_SIZE = (224,224)
DATASET_DIR = "dataset/cell_images"
PROCESSED_DATASET_DIR = "dataset/processed"
MODEL_PATH = "model/malaria_model.keras"

# Dataset prep
split_dataset()
train_folder = os.path.join(PROCESSED_DATASET_DIR, "train")
val_folder = os.path.join(PROCESSED_DATASET_DIR, "val")
test_folder = os.path.join(PROCESSED_DATASET_DIR, "test")

train_gen, val_gen, test_gen = create_data_generators(
    train_folder, val_folder, test_folder,
    batch_size=BATCH_SIZE,
    img_size=IMG_SIZE
)

print("Class indices:", train_gen.class_indices, flush=True)


# Model
model = Sequential([
    Input(shape=(*IMG_SIZE, 3)),
    
    # First Convolutional Block
    Conv2D(32, (3, 3), activation='relu'),
    MaxPooling2D(2, 2),
    BatchNormalization(),
    
    # Second Convolutional Block
    Conv2D(64, (3, 3), activation='relu'),
    MaxPooling2D(2, 2),
    BatchNormalization(),
    
    # Third Convolutional Block (New Layer)
    Conv2D(128, (3, 3), activation='relu'),
    MaxPooling2D(2, 2),
    BatchNormalization(),
    
    # Fourth Convolutional Block (New Layer)
    Conv2D(256, (3, 3), activation='relu'),
    MaxPooling2D(2, 2),
    BatchNormalization(),
    
    # Flatten the output for the Dense layers
    Flatten(),
    
    # Fully Connected Layer
    Dense(128, activation='relu'),
    Dropout(0.4),
    
    # Output layer for binary classification (infected or not)
    Dense(1, activation='sigmoid')
])

# Enhanced model compilation with additional metrics
model.compile(
    optimizer=Adam(learning_rate=0.0001), 
    loss='binary_crossentropy', 
    metrics=['accuracy', 'Precision', 'Recall', 'AUC']
)

# Save model summary to file
model.summary(print_fn=lambda x: open("model_summary.txt", "a", encoding="utf-8").write(x + "\n"))


# Calculate class weights for handling imbalance
train_labels = train_gen.classes
class_weights = compute_class_weight(
    class_weight='balanced',
    classes=np.unique(train_labels),
    y=train_labels
)
class_weight_dict = dict(enumerate(class_weights))

# Enhanced callbacks
checkpoint = ModelCheckpoint(MODEL_PATH, save_best_only=True)
early_stop = EarlyStopping(patience=10, restore_best_weights=True)
reduce_lr = ReduceLROnPlateau(monitor='val_loss', factor=0.2, patience=5, min_lr=1e-6, verbose=1)

# Train with improved parameters
history = model.fit(
    train_gen,
    validation_data=val_gen,
    epochs=EPOCHS,
    callbacks=[early_stop, checkpoint, reduce_lr],
    class_weight=class_weight_dict
)

print(f"✅ Best model saved at: {MODEL_PATH}")

# Evaluate
results = model.evaluate(val_gen)
loss, acc, precision, recall, auc = results
print(f"Validation Accuracy: {acc * 100:.2f}%")
print(f"Validation Precision: {precision * 100:.2f}%")
print(f"Validation Recall: {recall * 100:.2f}%")
print(f"Validation AUC: {auc:.4f}")

# Visuals
plt.figure(figsize=(12, 5))
plt.subplot(1, 2, 1)
plt.plot(history.history['accuracy'], label='Train Accuracy')
plt.plot(history.history['val_accuracy'], label='Validation Accuracy')
plt.legend()
plt.title('Model Accuracy')

plt.subplot(1, 2, 2)
plt.plot(history.history['loss'], label='Train Loss')
plt.plot(history.history['val_loss'], label='Validation Loss')
plt.legend()
plt.title('Model Loss')
plt.tight_layout()
plt.savefig('static/training_curves.png')
plt.show()

# Enhanced evaluation with confusion matrix
print("\nEvaluating on test data...")
test_gen.reset()
y_pred = model.predict(test_gen, steps=len(test_gen), verbose=1)
y_pred_binary = (y_pred > 0.5).astype(int)
y_true = test_gen.classes

# Plot confusion matrix
cm = confusion_matrix(y_true, y_pred_binary)
plt.figure(figsize=(8, 6))
sns.heatmap(cm, annot=True, fmt='d', cmap='Blues', 
            xticklabels=['Uninfected', 'Parasitized'],
            yticklabels=['Uninfected', 'Parasitized'])
plt.xlabel('Predicted')
plt.ylabel('True')
plt.title('Confusion Matrix')
plt.savefig('static/confusion_matrix.png')
plt.show()

# Print classification report
print("\nClassification Report:")
print(classification_report(y_true, y_pred_binary, 
                         target_names=['Uninfected', 'Parasitized']))

def generate_gradcam(model, img_path, output_path="static/gradcam.jpg"):
    img = cv2.imread(img_path)
    img = cv2.resize(img, (224, 224))
    img_array = np.expand_dims(img, axis=0) / 255.0

    grad_model = tf.keras.models.Model(
        [model.inputs], 
        [model.get_layer("conv2d_3").output, model.output]
    )

    with tf.GradientTape() as tape:
        conv_outputs, predictions = grad_model(img_array)
        loss = predictions[0]

    grads = tape.gradient(loss, conv_outputs)
    pooled_grads = tf.reduce_mean(grads, axis=(0, 1, 2))
    conv_outputs = conv_outputs[0]
    heatmap = conv_outputs @ pooled_grads[..., tf.newaxis]
    heatmap = tf.squeeze(heatmap)
    heatmap = np.maximum(heatmap, 0) / (tf.reduce_max(heatmap) + 1e-10)
    heatmap = cv2.resize(heatmap.numpy(), (224, 224))
    heatmap = np.uint8(255 * heatmap)

    heatmap_color = cv2.applyColorMap(heatmap, cv2.COLORMAP_JET)
    superimposed_img = cv2.addWeighted(img, 0.6, heatmap_color, 0.4, 0)

    cv2.imwrite(output_path, superimposed_img)
    return output_path



def is_valid_cell_image(image_path):
    import cv2

    img = cv2.imread(image_path)
    if img is None:
        return False

    # ✅ Relaxed dimension check
    h, w = img.shape[:2]
    if h < 40 or w < 40:
        return False

    # Convert to grayscale and blur
    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    blur = cv2.medianBlur(gray, 5)

    # Hough Circle detection (relax parameters)
    circles = cv2.HoughCircles(
        blur, cv2.HOUGH_GRADIENT, dp=1, minDist=10,
        param1=30, param2=20, minRadius=5, maxRadius=70
    )

    # 🧪 Debug Print (optional)
    # print(f"Circle detection: {circles}")

    return circles is not None and len(circles[0]) > 0



def predict_image(model, image_path):
    from tensorflow.keras.preprocessing import image
    import numpy as np
    import matplotlib.pyplot as plt

    img = image.load_img(image_path, target_size=(224, 224))
    img_array = image.img_to_array(img)
    img_array = np.expand_dims(img_array, axis=0)
    img_array = img_array / 255.0

    prediction = model.predict(img_array)[0][0]  # Single float

    # 🔁 Reverse the logic for testing (ONLY for debug)
    is_infected = prediction < 0.5
    confidence = 1 - prediction if is_infected else prediction

    print("Prediction Score:", prediction)
    print("Classified as:", "Infected" if is_infected else "Uninfected")
    print("Confidence:", confidence)

    # Debug: Visualize
    #plt.imshow(img_array[0])
    #plt.title(f"{'Infected' if is_infected else 'Uninfected'} ({confidence*100:.2f}%)")
    #plt.axis('off')
    #plt.show()

    return is_infected, round(float(confidence),4)

def preprocess_image(image_path, target_size=(224,224)):
    img = cv2.imread(image_path)
    if img is None:
        raise Exception("Cannot read image.")
    img = cv2.resize(img, target_size)
    img = cv2.cvtColor(img, cv2.COLOR_BGR2RGB)
    img = img / 255.0
    return np.expand_dims(img, axis=0)

def load_trained_model(model_path="model/malaria_model.keras"):
    if not os.path.exists(model_path):
        raise FileNotFoundError(f"Model not found at {model_path}")
    return tf.keras.models.load_model(model_path)

def allowed_file(filename):
    return '.' in filename and filename.rsplit('.', 1)[1].lower() in {'png', 'jpg', 'jpeg'}

def save_uploaded_file(file):
    upload_folder = "static/uploads"

def cleanup_old_files(folder="static/uploads", max_files=50):
    files = sorted([os.path.join(folder, f) for f in os.listdir(folder)], key=os.path.getmtime)
    for file in files[:-max_files]:
        try:
            os.remove(file)
        except:
            continue
